APK player

Two independent players on one page. Each APK runs in its own isolated frame with its own VM Worker; the attributes below are the same settings as the lab's "Dispositivo virtual" panel.

Embed it

<script type="module" src="https://YOUR-HOST/player/apk-player.js"></script>

<apk-player src="/apks/app-debug.apk" device="phone" locale="pt-BR"
            theme="dark" font-scale="1.1" orientation="portrait"
            style="height:640px"></apk-player>

<script type="module">
  const player = document.querySelector('apk-player');
  player.addEventListener('ready', e => console.log('running', e.detail.activity));
  player.addEventListener('error', e => console.warn(e.detail.report));   // cause, frames, stubbed APIs, log tail
  // tolerate-missing="primitives" (attribute, off by default) stubs missing Android APIs; await player.diagnostics() lists them
  player.setConfig({ theme: 'light', locale: 'en-US' });   // applied live (Android semantics)
</script>

Mount from a File or ArrayBuffer

import { ApkPlayer } from 'https://YOUR-HOST/player/apk-player.js';
const player = ApkPlayer.mount(document.querySelector('#slot'), {
  apk: fileInput.files[0],                 // File | Blob | ArrayBuffer (transferred) | URL string
  config: { device: 'tablet', theme: 'dark' },
  assetsBase: 'https://YOUR-HOST/'         // where player/frame.html and the runtime live
});
player.addEventListener('ready', e => console.log(e.detail));
player.destroy();
